1. Servidor Flask (app.py)

Recibe las selecciones del usuario desde la interfaz web y ejecuta la consulta lógica en la librería pyswip. 

from flask import Flask, request, jsonify
from pyswip import Prolog

app = Flask(__name__)
prolog = Prolog()
prolog.consult("recetas.pl") # Carga base de conocimientos

@app.route('/api/recomendar', methods=['POST'])
def recomendar():
    data = request.json
    ingrediente = data.get('ingrediente')
    categoria = data.get('categoria')
    
    # Construir consulta Prolog
    query = f"recomendar_por_categoria(Receta, {categoria}, {ingrediente})"
    resultados = list(prolog.query(query))
    
    recetas = [res['Receta'] for res in resultados]
    return jsonify({"recetas": recetas})

2. Regla Ejecutada en Prolog (recetas.pl)

Prolog recibe las variables instanciadas, realiza la unificación y evalúa el backtracking de las alternativas. 

% Regla invocada por la API web
recomendar_por_categoria(Receta, TipoDeseado, IngredienteUsuario) :-
    tipo_receta(Receta, TipoDeseado),
    recomendar_receta(Receta, IngredienteUsuario).

% Regla de sustitutos (evalúa alternativos)
recomendar_receta(Receta, IngredienteUsuario) :-
    ingrediente_receta(Receta, IngredienteRequerido),
    ( IngredienteRequerido == IngredienteUsuario 
    ; 
      ingrediente_alternativo(IngredienteRequerido, IngredienteUsuario) 
    ).

% --------------------------------------------------------------------
% 4. REGLAS LÓGICAS DEL SISTEMA EXPERTO (10 REGLAS NO TRIVIALES)
% --------------------------------------------------------------------

% Regla 1. Verificar si una receta pertenece a un momento del día
es_del_momento(Receta, Momento) :-
    tipo_receta(Receta, Momento).

% Regla 2. Disponibilidad e inferencia con sustitutos/alternativas
recomendar_receta(Receta, IngredienteUsuario) :-
    ingrediente_receta(Receta, IngredienteRequerido),
    ( IngredienteRequerido == IngredienteUsuario 
    ; 
      ingrediente_alternativo(IngredienteRequerido, IngredienteUsuario) 
    ).

% Regla 3. Recomendar por categoría general y disponibilidad
recomendar_por_categoria(Receta, TipoDeseado, IngredienteUsuario) :-
    tipo_receta(Receta, TipoDeseado),
    recomendar_receta(Receta, IngredienteUsuario).

% Regla 4. Filtrado específico para Desayuno
recomendar_desayuno(Receta, IngredienteUsuario) :-
    tipo_receta(Receta, desayuno),
    recomendar_receta(Receta, IngredienteUsuario).

% Regla 5. Filtrado específico para Comida
recomendar_comida(Receta, IngredienteUsuario) :-
    tipo_receta(Receta, comida),
    recomendar_receta(Receta, IngredienteUsuario).

% Regla 6. Filtrado específico para Cena
recomendar_cena(Receta, IngredienteUsuario) :-
    tipo_receta(Receta, cena),
    recomendar_receta(Receta, IngredienteUsuario).

% Regla 7. Seguridad Alimentaria: Exclusión de carnes (Vegetariano - Negación por falla)
receta_sin_carne(Receta) :-
    tipo_receta(Receta, _), 
    \+ ingrediente_receta(Receta, pollo_deshebrado),
    \+ ingrediente_receta(Receta, pepperoni).

% Regla 8. Seguridad Alimentaria: Exclusión de lácteos (\+ Negación por falla)
receta_sin_lacteos(Receta) :-
    tipo_receta(Receta, _),
    \+ ingrediente_receta(Receta, quesillo_deshebrado),
    \+ ingrediente_receta(Receta, crema).

% Regla 9. Recomendación combinada Vegetariano + Disponibilidad
recomendar_vegetariano(Receta, IngredienteUsuario) :-
    receta_sin_carne(Receta),
    recomendar_receta(Receta, IngredienteUsuario).

% Regla 10. Verificación directa de pertenencia de ingrediente
verificar_ingrediente_receta(Receta, Ingrediente) :-
    ingrediente_receta(Receta, Ingrediente).
